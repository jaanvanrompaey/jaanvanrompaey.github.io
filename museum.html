function updateStatus(msg, size = null) { statusEl.textContent = msg + (size !== null ? ` | ${size} bytes` : ''); } function showDownload(blob, filename) { const url = URL.createObjectURL(blob); downloadLink.href = url; downloadLink.download = filename; downloadLink.classList.remove('hidden'); } // Core Logic fileInput.addEventListener('change', async (e) => { const file = e.target.files[0]; if (!file) return; // Reset UI downloadLink.classList.add('hidden'); updateStatus(`Processing: ${file.name} (${(file.size/1024).toFixed(1)} KB)`); const options = { targetSizeBytes: parseInt(document.getElementById('targetSize').value) || 512, qualityStep: parseInt(document.getElementById('qualityStep').value) || 10, dimensionStep: parseFloat(document.getElementById('dimStep').value) || 0.1, maxIterations: 20, stallThreshold: 50, maxStalls: 5 }; const result = await optimizeImage(file, options); if (result) { showDownload(result.blob, result.name); } else { updateStatus(`⚠️ Warning: Could not reach ≤${options.targetSizeBytes} bytes within limits.`); } }); async function optimizeImage(file, options) { const img = await loadImage(file); const baseName = file.name.replace(/\.[^/.]+$/, ""); const outExt = '.jpg'; let prevFileSize = Infinity; let stallCount = 0; for (let i = 1; i <= options.maxIterations; i++) { const reductionFactor = Math.max(0.0, 1.0 - (i * options.dimensionStep)); const newW = Math.max(1, Math.floor(img.width * reductionFactor)); const newH = Math.max(1, Math.floor(img.height * reductionFactor)); const currentQuality = Math.max(1, 100 - (i * options.qualityStep)); const outputName = `${baseName}_v${i}${outExt}`; updateStatus(`[${i}] ${newW}x${newH} | Q:${currentQuality}`); const blob = await canvasToBlob(img, newW, newH, currentQuality); const fileSize = blob.size; updateStatus(`[${i}] ${newW}x${newH} | Q:${currentQuality} | ${fileSize} bytes`, fileSize); if (fileSize <= options.targetSizeBytes) { updateStatus(`✅ Success! Final file: ${outputName} (${fileSize} bytes)`); return { blob, name: outputName, size: fileSize }; } if (Math.abs(fileSize - prevFileSize) < options.stallThreshold) { stallCount++; if (stallCount >= options.maxStalls) { updateStatus(`⏹️ Linear optimization stalled. Returning best available: ${outputName}`); return { blob, name: outputName, size: fileSize }; } } else { stallCount = 0; } prevFileSize = fileSize; } updateStatus(`⚠️ Warning: Could not reach ≤${options.targetSizeBytes} bytes.`); return null; } // Browser API Helpers function loadImage(file) { return new Promise((resolve, reject) => { const img = new Image(); img.onload = () => resolve(img); img.onerror = reject; img.src = URL.createObjectURL(file); }); } function canvasToBlob(img, w, h, quality) { return new Promise((resolve) => { const canvas = document.createElement('canvas'); canvas.width = w; canvas.height = h; const ctx = canvas.getContext('2d'); // JPEG doesn't support transparency. Fill with white background. ctx.fillStyle = '#FFFFFF'; ctx.fillRect(0, 0, w, h); ctx.drawImage(img, 0, 0, w, h); // quality is 0-1 for toBlob, Python uses 0-100 canvas.toBlob((blob) => resolve(blob), 'image/jpeg', quality / 100); }); }
